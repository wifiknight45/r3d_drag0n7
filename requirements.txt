# requirements.txt
# Python dependencies for sistema_completa.py
# Astrological Chart Generator
# Requires Python 3.9+ (or 3.8 with backports.zoneinfo)

# ============================================================================
# CORE ASTRONOMICAL/ASTROLOGICAL LIBRARIES
# ============================================================================

# Swiss Ephemeris - High precision ephemeris calculations (Nostradamus mode)
# Provides exact planetary positions and house calculations
# Note: Requires ~100MB ephemeris data files (separate download)
pyswisseph>=2.10.3.2

# Skyfield - JPL ephemeris for astronomical calculations (Ptolemy mode fallback)
# Pure Python alternative when Swiss Ephemeris unavailable
# First run downloads DE421 ephemeris (~17MB)
skyfield>=1.46

# ============================================================================
# GEOLOCATION AND TIMEZONE
# ============================================================================

# Geopy - Geocoding service to convert locations to coordinates
# Uses Nominatim (OpenStreetMap) for location lookups

# TimezoneFinder - Convert lat/lon coordinates to timezone names
# Essential for accurate time calculations
timezonefinder>=6.2.0

# Numba - JIT compiler for performance boost (optional but recommended)
# Significantly speeds up timezonefinder lookups

# ============================================================================
# DATE/TIME PARSING AND TIMEZONE SUPPORT
# ============================================================================

# Python-dateutil - Flexible date/time parsing
# Handles various date formats and timezone conversions

# Backports of zoneinfo for Python < 3.9
# Required for Python 3.8 and earlier

# ============================================================================
# WEB REQUESTS AND API ACCESS
# ============================================================================

#   pip install -r requirements.txt

#   pip install pyswisseph skyfield python-dateutil requests geopy timezonefinder

# ============================================================================
# MODE-SPECIFIC REQUIREMENTS
# ============================================================================

# NOSTRADAMUS MODE (highest accuracy):
#   Requires: pyswisseph + Swiss Ephemeris data files
#   Download ephemeris files from: https://www.astro.com/ftp/swisseph/ephe/
#   Required files: seas_18.se1, semo_18.se1, sepl_18.se1 (or newer)
#   Place in: data/ directory
#   Total size: ~100MB

# PTOLEMY MODE (fallback, still accurate):
#   Requires: skyfield (auto-downloads JPL DE421 ephemeris on first run)
#   Initial download: ~17MB
#   No manual ephemeris setup needed

# PARANOID MODE (completely offline):
#   Requires: Pre-downloaded ephemeris files + data/zipcodes.csv
#   Setup: Download all files before use, disable internet access
#   Uses: Local geocoding and cached event data

# ============================================================================
# TROUBLESHOOTING
# ============================================================================

# Problem: pyswisseph won't compile
# Solution 1: pip install --upgrade pip setuptools wheel
# Solution 2: pip install pyswisseph --only-binary :all:
# Solution 3: Use Ptolemy mode instead (still very accurate)

# Problem: timezonefinder is slow
# Solution: pip install numba (enables JIT compilation)

# Problem: Import errors after installation
# Solution: python -m pip install --force-reinstall -r requirements.txt

# Problem: PDF generation fails with encoding errors
# Solution: Ensure you're using fpdf2, not fpdf 1.7.2

# Problem: Skyfield can't download ephemeris
# Solution: Pre-download DE421 from https://naif.jpl.nasa.gov/pub/naif/generic_kernels/spk/planets/
#           Place in ~/.local/share/skyfield/ (Linux/Mac) or %LOCALAPPDATA%\skyfield\ (Windows)
